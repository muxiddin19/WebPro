useEffect syntax: mount and cleanup

Source 01-EffectSyntax.jsx

import { useState, useEffect } from 'react';

function MyComponent() {
  const [data, setData] = useState(null);

  useEffect(() => {
    // Runs after the first render (empty dependency array)
    console.log('Component rendered!');

    // Optional: return a cleanup function
    return () => {
      console.log('Cleanup before next run or unmount');
    };
  }, []);

  return <div>{JSON.stringify(data)}</div>;
}

// Demo wrapper: mount and unmount the component
export default function Demo() {
  const [shown, setShown] = useState(true);
  return (
    <div>
      <button onClick={() => setShown(s => !s)}>{shown ? 'Unmount' : 'Mount'}</button>
      {shown && <MyComponent />}
    </div>
  );
}

The three dependency patterns

Compare which effects log when you change val and when you only re-render.

Live result

Source 02-DependencyPatterns.jsx

import { useState, useEffect } from 'react';

function RunsOnce() {
  useEffect(() => { console.log('[]     -> ran once after mount'); }, []);
  return <p>[] : once after mount</p>;
}

function RunsOnChange({ val }) {
  useEffect(() => { console.log('[val]  -> ran, val =', val); }, [val]);
  return <p>[val] : when val changes</p>;
}

function RunsEveryRender() {
  // No dependency array: usually a bug
  useEffect(() => { console.log('(none) -> ran after EVERY render'); });
  return <p>(none) : after every render</p>;
}

export default function DependencyPatterns() {
  const [val, setVal]   = useState(0);
  const [tick, setTick] = useState(0);
  return (
    <div>
      <button onClick={() => setVal(v => v + 1)}>val++ (changes val)</button>
      <button onClick={() => setTick(t => t + 1)}>re-render (val unchanged)</button>
      <p>val = {val}, extra renders = {tick}</p>
      <RunsOnce />
      <RunsOnChange val={val} />
      <RunsEveryRender />
    </div>
  );
}

Source 03-UserProfile.jsx

import { useState, useEffect } from 'react';

export function UserProfile({ userId }) {
  const [user, setUser]       = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);

  useEffect(() => {
    setLoading(true);
    setError(null);
    fetch(`https://api.example.com/users/${userId}`)
      .then(res => {
        if (!res.ok) throw new Error('User not found');
        return res.json();
      })
      .then(data => { setUser(data); setLoading(false); })
      .catch(err  => { setError(err.message); setLoading(false); });
  }, [userId]);       // re-fetch when userId prop changes

  if (loading) return <p>Loading...</p>;
  if (error)   return <p>Error: {error}</p>;
  return <h2>{user?.name}</h2>;
}

// Demo wrapper: change the userId prop
export default function Demo() {
  const [id, setId] = useState(1);
  return (
    <div>
      {[1, 2, 3, 99].map(n => (
        <button key={n} onClick={() => setId(n)}>userId {n}</button>
      ))}
      <UserProfile userId={id} />
    </div>
  );
}

async/await inside an effect

The async function is defined inside the effect and called immediately.

Live result

Source 04-AsyncEffect.jsx

import { useState, useEffect } from 'react';

export default function AsyncEffect() {
  const [user, setUser]       = useState(null);
  const [error, setError]     = useState(null);
  const [loading, setLoading] = useState(true);

  // WRONG: useEffect(async () => { ... }, [])
  //   an async callback returns a Promise, but useEffect
  //   expects undefined or a cleanup function

  // CORRECT: define the async function inside and call it
  useEffect(() => {
    async function fetchData() {
      try {
        const res  = await fetch('https://api.github.com/users/muxiddin19');
        const data = await res.json();
        setUser(data);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    }
    fetchData();
    // No return needed here unless we have cleanup
  }, []);

  if (loading) return <p>Loading...</p>;
  if (error)   return <p>Error: {error}</p>;
  return <p>{user.login}: {user.name}</p>;
}

Cleanup: timer

Mount LeakyClock, then Unmount: the console keeps logging ticks. LiveClock stops cleanly.

Live result

Source 05-LiveClock.jsx

import { useState, useEffect } from 'react';

function LiveClock() {
  const [time, setTime] = useState(new Date());

  useEffect(() => {
    const intervalId = setInterval(() => {
      setTime(new Date());
    }, 1000);

    // CORRECT: return cleanup, stop the timer when the component unmounts
    return () => clearInterval(intervalId);
  }, []);    // empty: start once on mount, cleanup on unmount

  return <p>{time.toLocaleTimeString('ko-KR')}</p>;
}

// WRONG: no cleanup. The interval keeps running after unmount.
// (This copy stops itself after 10 ticks so the demo stays safe.)
function LeakyClock() {
  const [time, setTime] = useState(new Date());

  useEffect(() => {
    let n = 0;
    const id = setInterval(() => {
      console.log('leaky tick #' + (++n) + ' (component may already be gone)');
      setTime(new Date());
      if (n >= 10) clearInterval(id);
    }, 1000);
  }, []);

  return <p>{time.toLocaleTimeString('ko-KR')}</p>;
}

// Demo wrapper
export default function Demo() {
  const [mode, setMode] = useState('none');
  return (
    <div>
      <button onClick={() => setMode('good')}>Mount LiveClock</button>
      <button onClick={() => setMode('leaky')}>Mount LeakyClock</button>
      <button onClick={() => setMode('none')}>Unmount</button>
      {mode === 'good'  && <LiveClock />}
      {mode === 'leaky' && <LeakyClock />}
    </div>
  );
}

Cleanup: AbortController

Type quickly in the box and watch the console: earlier requests are aborted.

Live result

Source 06-AbortFetch.jsx

import { useState, useEffect } from 'react';

export function UserCard({ username }) {
  const [user, setUser]   = useState(null);
  const [error, setError] = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    setError(null);

    async function fetchUser() {
      try {
        const res = await fetch(
          `https://api.github.com/users/${username}`,
          { signal: controller.signal }   // link to controller
        );
        if (!res.ok) throw new Error('User not found');
        const data = await res.json();
        setUser(data);
      } catch (err) {
        if (err.name === 'AbortError') {            // ignore cancelled requests
          console.log('aborted request for', username);
          return;
        }
        setError(err.message);
      }
    }
    fetchUser();

    // Cancel the in-flight request if username changes
    return () => controller.abort();
  }, [username]);

  if (error) return <p>Error: {error}</p>;
  return <p>{user ? user.login + ' / ' + user.name : 'Loading...'}</p>;
}

// Demo wrapper: type to change the username
export default function Demo() {
  const [name, setName] = useState('react-learner');
  return (
    <div>
      <input value={name} onChange={e => setName(e.target.value)} />
      <UserCard username={name} />
    </div>
  );
}

import { useState, useEffect } from 'react';

// Code challenge: one possible solution
export default function GitHubUserLookup() {
  const [input, setInput]       = useState('');
  const [username, setUsername] = useState('');   // set when Search is clicked
  const [user, setUser]         = useState(null);
  const [loading, setLoading]   = useState(false);
  const [error, setError]       = useState(null);

  useEffect(() => {
    if (!username) return;
    const controller = new AbortController();   // BONUS: abort previous request
    setLoading(true); setError(null); setUser(null);

    fetch(`https://api.github.com/users/${username}`, { signal: controller.signal })
      .then(res => {
        if (res.status === 404) throw new Error('User not found (404)');
        if (!res.ok) throw new Error('Request failed: ' + res.status);
        return res.json();
      })
      .then(data => { setUser(data); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [username]);

  return (
    <div>
      <input value={input} placeholder="GitHub username"
             onChange={e => setInput(e.target.value)} />
      <button onClick={() => setUsername(input.trim())}>Search</button>

      {loading && <p>Loading...</p>}
      {error && <p style={{ color: 'crimson' }}>{error}</p>}
      {user && (
        <div>
          <img src={user.avatar_url} alt="" width="64" height="64" />
          <h3>{user.name || user.login}</h3>
          <p>{user.bio}</p>
          <p>{user.followers} followers · {user.public_repos} public repos</p>
          <a href={user.html_url}>{user.html_url}</a>
        </div>
      )}
    </div>
  );
}

Race condition and the fix

Type “abcdef” quickly with the checkbox off, then on. Compare the result shown.

Live result

Source 12-RaceCondition.jsx

import { useState, useEffect } from 'react';

// In this demo, requests for SHORTER names are slower, so responses
// arrive out of order when you type quickly.
export default function RaceCondition() {
  const [query, setQuery]   = useState('');
  const [result, setResult] = useState('');
  const [safe, setSafe]     = useState(false);

  useEffect(() => {
    if (!query) return;
    const controller = new AbortController();

    fetch(`https://api.github.com/users/${query}`,
          { signal: safe ? controller.signal : undefined })
      .then(res => res.json())
      .then(data => setResult(data.login))
      .catch(err => { if (err.name !== 'AbortError') console.log(err.message); });

    // SAFE mode: cancel the previous request when query changes
    return safe ? () => controller.abort() : undefined;
  }, [query, safe]);

  return (
    <div>
      <label>
        <input type="checkbox" checked={safe} onChange={e => setSafe(e.target.checked)} />
        {' '}Abort previous request (the fix)
      </label>
      <p><input value={query} placeholder="type slowly, then quickly: abcdef"
                onChange={e => setQuery(e.target.value)} /></p>
      <p>You typed: <b>{query || '-'}</b></p>
      <p>Result shown for: <b style={{ color: result && result !== query ? 'crimson' : 'inherit' }}>
        {result || '-'}</b> {result && result !== query && ' (stale!)'}</p>
    </div>
  );
}

localStorage persistence

Toggle the theme, then reload the page: the choice is remembered (this page’s own theme changes too).

Live result

Source 08-ThemeToggle.jsx

import { useState, useEffect } from 'react';

export default function ThemeToggle() {
  // Read once at mount (lazy initializer form)
  const [theme, setTheme] = useState(
    () => localStorage.getItem('theme') || 'light'
  );

  // Write whenever theme changes
  useEffect(() => {
    localStorage.setItem('theme', theme);
    document.documentElement.setAttribute('data-theme', theme);
  }, [theme]);

  return (
    <button onClick={() => setTheme(t => t === 'light' ? 'dark' : 'light')}>
      Current: {theme} — Click to toggle
    </button>
  );
}

Gotcha: infinite loop

BadLoop is stopped by a safety guard after 8 runs. FixedOnce runs a single time.

Live result

Source 10-InfiniteLoopGuarded.jsx

import { useState, useEffect, useRef } from 'react';

// BAD: setState inside the effect + that same state in the dependency array
function BadLoop() {
  const [count, setCount] = useState(0);
  const runs = useRef(0);

  useEffect(() => {
    runs.current += 1;
    // Safety guard so this demo cannot freeze the page
    if (runs.current > 8) { console.log('safety stop: this loop would never end'); return; }
    console.log('effect run #' + runs.current + ', count =', count);
    setCount(count + 1);          // changes count -> effect runs again
  }, [count]);

  return <p>BadLoop count = {count}</p>;
}

// FIX: functional update with an empty dependency array
function FixedOnce() {
  const [count, setCount] = useState(0);

  useEffect(() => {
    setCount(c => c + 1);
    console.log('FixedOnce effect ran');
  }, []);

  return <p>FixedOnce count = {count}</p>;
}

// Demo wrapper
export default function Demo() {
  const [which, setWhich] = useState('none');
  return (
    <div>
      <button onClick={() => setWhich('bad')}>Run BadLoop</button>
      <button onClick={() => setWhich('fixed')}>Run FixedOnce</button>
      <button onClick={() => setWhich('none')}>Unmount</button>
      {which === 'bad'   && <BadLoop />}
      {which === 'fixed' && <FixedOnce />}
    </div>
  );
}

You might not need an effect

Type in either box and compare the render counters.

Live result

Source 11-DerivedState.jsx

import { useState, useEffect, useRef } from 'react';

// Unnecessary effect: extra state and an extra render
function WithEffect({ first, last }) {
  const renders = useRef(0); renders.current += 1;   // demo counter only
  const [fullName, setFullName] = useState('');
  useEffect(() => { setFullName(first + ' ' + last); }, [first, last]);
  return <p>WithEffect: {fullName} (renders: {renders.current})</p>;
}

// Better: derive the value while rendering
function Derived({ first, last }) {
  const renders = useRef(0); renders.current += 1;   // demo counter only
  const fullName = first + ' ' + last;
  return <p>Derived: {fullName} (renders: {renders.current})</p>;
}

export default function Demo() {
  const [first, setFirst] = useState('Kim');
  const [last, setLast]   = useState('Minho');
  return (
    <div>
      <input value={first} onChange={e => setFirst(e.target.value)} />
      <input value={last}  onChange={e => setLast(e.target.value)} />
      <WithEffect first={first} last={last} />
      <Derived first={first} last={last} />
    </div>
  );
}

import { useState, useEffect } from 'react';

// Homework starter: Weather Dashboard (Open-Meteo, no API key)
// TODO 1: let the user type a CITY name and look up coordinates
//         (https://geocoding-api.open-meteo.com/v1/search?name=Seoul)
// TODO 2: map weather_code to a description ("Clear", "Rain", ...)
// TODO 3: BONUS: use navigator.geolocation for the first coordinates
export default function WeatherStarter() {
  const [lat, setLat]         = useState(37.46);
  const [lon, setLon]         = useState(127.01);
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true); setError(null);

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${lat}&longitude=${lon}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('Could not load weather (check coordinates)');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [lat, lon]);

  return (
    <div>
      <label>lat <input type="number" value={lat} onChange={e => setLat(e.target.value)} /></label>
      <label> lon <input type="number" value={lon} onChange={e => setLon(e.target.value)} /></label>
      {loading && <p>Loading...</p>}
      {error && <p style={{ color: 'crimson' }}>{error}</p>}
      {weather && !loading && !error && (
        <p>{weather.temperature_2m} °C · wind {weather.wind_speed_10m} km/h · code {weather.weather_code}</p>
      )}
    </div>
  );
}
