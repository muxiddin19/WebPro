useState and functional updates

Try “+3 (stale)” versus “+3 (functional)”. Same code idea, different result.

Live result

Source 02-CounterState.jsx

import { useState } from 'react';

// useState + functional updates
export default function Counter() {
  const [count, setCount] = useState(0);

  const increment = () => setCount(prev => prev + 1);
  const decrement = () => setCount(prev => prev - 1);
  const reset     = () => setCount(0);

  // Stale: all three calls read the same old count
  const addThreeStale = () => {
    setCount(count + 1); setCount(count + 1); setCount(count + 1);
  };
  // Safe: each call gets the latest value
  const addThreeSafe = () => {
    setCount(p => p + 1); setCount(p => p + 1); setCount(p => p + 1);
  };

  return (
    <div>
      <button onClick={decrement}>-</button>
      <span> {count} </span>
      <button onClick={increment}>+</button>
      <button onClick={reset}>Reset</button>
      <p>
        <button onClick={addThreeStale}>+3 (stale)</button>
        <button onClick={addThreeSafe}>+3 (functional)</button>
      </p>
    </div>
  );
}

Multiple useState calls

Four independent pieces of state. Fill in the form and submit.

Live result

Source 03-UserForm.jsx

import { useState } from 'react';

export default function UserForm() {
  const [name,      setName]      = useState('');
  const [email,     setEmail]     = useState('');
  const [age,       setAge]       = useState(0);
  const [submitted, setSubmitted] = useState(false);

  function handleSubmit(e) {
    e.preventDefault();
    setSubmitted(true);
  }

  if (submitted) {
    return <p>Thanks, {name}! Email: {email}, Age: {age}</p>;
  }

  return (
    <form onSubmit={handleSubmit}>
      <input value={name}  onChange={e => setName(e.target.value)}  placeholder="Name" />
      <input value={email} onChange={e => setEmail(e.target.value)} placeholder="Email" />
      <input type="number" value={age} onChange={e => setAge(e.target.value)} />
      <button>Submit</button>
    </form>
  );
}

Source 04-ProfileForm.jsx

import { useState } from 'react';

export default function ProfileForm() {
  const [user, setUser] = useState({ name: '', email: '', bio: '' });

  // WRONG: user.name = 'Kim Minho'; setUser(user);  (same object)
  // CORRECT: spread creates a new object
  const handleChange = (field, value) => {
    setUser(prev => ({ ...prev, [field]: value }));
  };

  return (
    <form>
      <input value={user.name}  onChange={e => handleChange('name',  e.target.value)} />
      <input value={user.email} onChange={e => handleChange('email', e.target.value)} />
      <textarea value={user.bio} onChange={e => handleChange('bio', e.target.value)} />
      <pre>{JSON.stringify(user, null, 2)}</pre>
    </form>
  );
}

State with arrays

Add, toggle (click the text) and remove items.

Live result

Source 05-TodoList.jsx

import { useState } from 'react';

export default function TodoList() {
  const [todos, setTodos] = useState([]);
  const [input, setInput] = useState('');

  // Add: spread existing + new item
  const addTodo = () => {
    if (!input.trim()) return;
    setTodos(prev => [...prev, { id: Date.now(), text: input, done: false }]);
    setInput('');
  };
  // Remove: filter creates a new array
  const removeTodo = (id) => setTodos(prev => prev.filter(t => t.id !== id));
  // Toggle: map creates a new array
  const toggleTodo = (id) =>
    setTodos(prev => prev.map(t => t.id === id ? { ...t, done: !t.done } : t));

  return (
    <div>
      <input value={input} onChange={e => setInput(e.target.value)} />
      <button onClick={addTodo}>Add</button>
      <ul>
        {todos.map(t => (
          <li key={t.id}>
            <span onClick={() => toggleTodo(t.id)}
                  style={{ textDecoration: t.done ? 'line-through' : 'none' }}>
              {t.text}
            </span>
            <button onClick={() => removeTodo(t.id)}>x</button>
          </li>
        ))}
      </ul>
    </div>
  );
}

Lifting state up

The parent holds the query. SearchBar writes it, ResultList reads it.

Live result

Source 06-SearchPage.jsx

import { useState } from 'react';

const FRUITS = ['Apple', 'Apricot', 'Banana', 'Cherry', 'Grape', 'Mango', 'Melon'];

// Parent owns the state
export default function SearchPage() {
  const [query, setQuery] = useState('');
  const results = FRUITS.filter(f => f.toLowerCase().includes(query.toLowerCase()));

  return (
    <>
      {/* Child 1 triggers updates via callback */}
      <SearchBar onSearch={setQuery} />
      {/* Child 2 reads state via props */}
      <ResultList query={query} results={results} />
    </>
  );
}

function SearchBar({ onSearch }) {
  const [input, setInput] = useState('');
  return <input value={input} placeholder="Search fruit" onChange={e => {
    setInput(e.target.value);
    onSearch(e.target.value);
  }} />;
}

function ResultList({ query, results }) {
  return (
    <div>
      <p>{results.length} result(s) for "{query}"</p>
      <ul>{results.map(r => <li key={r}>{r}</li>)}</ul>
    </div>
  );
}

Code challenge: counter with history

A possible solution. Use Undo to step back through the last 5 values.

Live result

Source 07-CounterWithHistory.jsx

import { useState } from 'react';

// Code challenge solution: two state variables, count + history
export default function CounterWithHistory() {
  const [count, setCount]     = useState(0);
  const [history, setHistory] = useState([0]);

  function change(next) {
    setCount(next);
    setHistory(h => [...h, next].slice(-5)); // keep last 5 values
  }

  function undo() {
    if (history.length < 2) return;
    setCount(history[history.length - 2]);
    setHistory(h => h.slice(0, -1));
  }

  return (
    <div>
      <h2>{count}</h2>
      <button onClick={() => change(count + 1)}>+1</button>
      <button onClick={() => change(count - 1)}>-1</button>
      <button onClick={() => change(0)}>Reset</button>
      <button onClick={undo}>Undo</button>
      <p>History: [{history.join(', ')}]</p>
    </div>
  );
}
