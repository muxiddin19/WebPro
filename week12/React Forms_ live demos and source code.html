import { useState, useRef } from 'react';

export default function ControlledVsUncontrolled() {
  const [text, setText] = useState('');
  const [read, setRead] = useState('');
  const inputRef = useRef(null);

  return (
    <div>
      <h4>1. Controlled: React state IS the value</h4>
      <input value={text} onChange={e => setText(e.target.value)} placeholder="type here" />
      <p>state = "{text}"</p>

      <h4>2. Controlled but with NO onChange: stuck</h4>
      {/* React warns in the console: value without onChange */}
      <input value="I cannot be edited" />

      <h4>3. Uncontrolled: the DOM owns the value</h4>
      <input ref={inputRef} defaultValue="edit me, then press the button" />
      <button onClick={() => setRead(inputRef.current.value)}>Read value (ref)</button>
      <p>read from the DOM: "{read}"</p>
    </div>
  );
}

import { useState } from 'react';

export default function PreferenceForm() {
  const [country, setCountry] = useState('KR');
  const [bio, setBio]         = useState('');
  const [agree, setAgree]     = useState(false);
  const [plan, setPlan]       = useState('free');

  return (
    <form>
      {/* Select: value on <select>, NOT on <option> */}
      <select value={country} onChange={e => setCountry(e.target.value)}>
        <option value="KR">South Korea</option>
        <option value="UZ">Uzbekistan</option>
        <option value="US">United States</option>
      </select>

      {/* Textarea: same pattern as <input> */}
      <p><textarea value={bio} onChange={e => setBio(e.target.value)} rows={3} /></p>

      {/* Checkbox: use checked, not value */}
      <label>
        <input type="checkbox" checked={agree}
               onChange={e => setAgree(e.target.checked)} /> I agree
      </label>

      {/* Radio: checked is true for the input whose value === state */}
      <p>
        <label><input type="radio" value="free" checked={plan === 'free'}
                      onChange={e => setPlan(e.target.value)} /> Free</label>
        <label><input type="radio" value="pro" checked={plan === 'pro'}
                      onChange={e => setPlan(e.target.value)} /> Pro</label>
      </p>

      <pre>{JSON.stringify({ country, bio, agree, plan }, null, 2)}</pre>
    </form>
  );
}

One handler for all fields

A single handleChange reads e.target.name. Note the checkbox branch.

Live result

Source 04-RegistrationForm.jsx

import { useState } from 'react';

export default function RegistrationForm() {
  const [form, setForm] = useState({
    name: '', email: '', password: '', role: 'student', subscribe: false
  });

  // One handler for ALL fields: reads e.target.name
  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setForm(prev => ({
      ...prev,
      [name]: type === 'checkbox' ? checked : value   // computed key
    }));
  };

  return (
    <form>
      <input name="name"     value={form.name}     onChange={handleChange} placeholder="Name" />
      <input name="email"    value={form.email}    onChange={handleChange} type="email" placeholder="Email" />
      <input name="password" value={form.password} onChange={handleChange} type="password" placeholder="Password" />
      <select name="role"    value={form.role}     onChange={handleChange}>
        <option value="student">Student</option>
        <option value="ta">Teaching Assistant</option>
      </select>
      {/* extra: shows the type === 'checkbox' branch */}
      <label>
        <input name="subscribe" type="checkbox" checked={form.subscribe} onChange={handleChange} /> Newsletter
      </label>
      <pre>{JSON.stringify(form, null, 2)}</pre>
    </form>
  );
}

import { useState } from 'react';

export default function LoginForm() {
  const [email, setEmail]       = useState('');
  const [password, setPassword] = useState('');
  const [status, setStatus]     = useState('idle'); // idle|loading|success|error

  const handleSubmit = async (e) => {
    e.preventDefault();   // stop the browser's default page reload
    setStatus('loading');
    try {
      const res = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password })
      });
      if (!res.ok) throw new Error('Invalid credentials');
      setStatus('success');
    } catch (err) { setStatus('error'); }
  };

  return (
    <form onSubmit={handleSubmit}>
      <p>Hint (simulated server): student@gachon.ac.kr / react1234</p>
      <input value={email}    onChange={e => setEmail(e.target.value)}    type="email" placeholder="Email" />
      <input value={password} onChange={e => setPassword(e.target.value)} type="password" placeholder="Password" />
      <button disabled={status === 'loading'}>
        {status === 'loading' ? 'Logging in...' : 'Log In'}
      </button>
      {status === 'error' && <p>Login failed. Check your credentials.</p>}
      {status === 'success' && <p>Welcome back!</p>}
      <p>status = {status}</p>
    </form>
  );
}

import { useState } from 'react';

export default function RegisterForm() {
  const [form, setForm]     = useState({ name: '', email: '', password: '' });
  const [errors, setErrors] = useState({});
  const [done, setDone]     = useState(false);

  const handleChange = (e) =>
    setForm(prev => ({ ...prev, [e.target.name]: e.target.value }));

  const validate = () => {
    const errs = {};
    if (!form.name.trim())
      errs.name = 'Name is required';
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email))
      errs.email = 'Enter a valid email address';
    if (form.password.length < 8)
      errs.password = 'Password must be at least 8 characters';
    return errs;    // empty object = all valid
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    const errs = validate();
    if (Object.keys(errs).length > 0) { setErrors(errs); return; }
    setErrors({});          // clear errors on a successful resubmit
    setDone(true);          // all valid: proceed with submission
  };

  if (done) return <p>Registered: {form.name} ({form.email})</p>;

  return (
    <form onSubmit={handleSubmit} noValidate>
      <p><input name="name" value={form.name} onChange={handleChange} placeholder="Name" />
         {errors.name && <span style={{ color: 'crimson' }}> {errors.name}</span>}</p>
      <p><input name="email" value={form.email} onChange={handleChange} placeholder="Email" />
         {errors.email && <span style={{ color: 'crimson' }}> {errors.email}</span>}</p>
      <p><input name="password" type="password" value={form.password} onChange={handleChange} placeholder="Password" />
         {errors.password && <span style={{ color: 'crimson' }}> {errors.password}</span>}</p>
      <button>Register</button>
    </form>
  );
}

import { useState, useRef } from 'react';

function SearchFocus() {
  const inputRef = useRef(null);

  const handleFocus = () => {
    inputRef.current.focus();    // focus the DOM element
    inputRef.current.select();   // also select existing text
  };

  return (
    <div>
      <input ref={inputRef} type="text" defaultValue="select me" />
      <button onClick={handleFocus}>Focus Search</button>
    </div>
  );
}

// File inputs MUST use useRef (they cannot be controlled)
function FileUpload() {
  const fileRef = useRef(null);
  const [info, setInfo] = useState('no file chosen');

  const handleUpload = () => {
    const file = fileRef.current.files[0];
    console.log(file.name, file.size, file.type);
    setInfo(file.name + ' · ' + file.size + ' bytes · ' + (file.type || 'unknown type'));
  };
  return (
    <div>
      <input type="file" ref={fileRef} onChange={handleUpload} />
      <p>{info}</p>
    </div>
  );
}

export default function Demo() {
  return (<div><h4>Focus management</h4><SearchFocus /><h4>File input</h4><FileUpload /></div>);
}

Code challenge: multi-step registration (solution)

Next stays disabled until the step is valid. Previous keeps your data. Submit logs the object.

Source 08-MultiStepRegistration.jsx

import { useState } from 'react';

const emailRe = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

// Validation is DERIVED from state: no extra state to keep in sync
function validateStep(step, f) {
  const e = {};
  if (step === 1) {
    if (f.name.trim().length < 2) e.name = 'At least 2 characters';
    if (!emailRe.test(f.email))   e.email = 'Enter a valid email address';
  }
  if (step === 2) {
    if (f.username.length < 3 || /\s/.test(f.username)) e.username = 'At least 3 characters, no spaces';
    if (f.password.length < 8)                           e.password = 'At least 8 characters';
    if (!f.confirm || f.confirm !== f.password)          e.confirm  = 'Passwords must match';
  }
  return e;
}

function Field({ label, name, type = 'text', value, onChange, onBlur, error }) {
  const id = 'f-' + name;
  return (
    <p>
      <label htmlFor={id}>{label} </label>
      <input id={id} name={name} type={type} value={value}
             onChange={onChange} onBlur={onBlur}
             aria-invalid={!!error} aria-describedby={error ? id + '-err' : undefined} />
      {error && <span id={id + '-err'} style={{ color: 'crimson' }}> {error}</span>}
    </p>
  );
}

export default function MultiStepRegistration() {
  const [step, setStep]       = useState(1);
  const [done, setDone]       = useState(false);
  const [touched, setTouched] = useState({});
  const [form, setForm] = useState({
    name: '', email: '', username: '', password: '', confirm: '',
    role: 'student', subscribe: false
  });

  const errors = validateStep(step, form);
  const valid  = Object.keys(errors).length === 0;

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setForm(prev => ({ ...prev, [name]: type === 'checkbox' ? checked : value }));
  };
  const handleBlur = (e) => setTouched(t => ({ ...t, [e.target.name]: true }));
  const shown = (name) => (touched[name] ? errors[name] : undefined);

  const next = () => { if (valid) setStep(s => s + 1); };
  const prev = () => setStep(s => s - 1);      // form data stays in state

  const handleSubmit = (e) => {
    e.preventDefault();
    const { confirm, ...fullFormObject } = form;
    console.log(fullFormObject);
    setDone(true);
  };

  if (done) return <p>Registration submitted. Check the console for the form object.</p>;

  const props = (name) => ({ name, value: form[name], onChange: handleChange, onBlur: handleBlur, error: shown(name) });

  return (
    <form onSubmit={handleSubmit}>
      <p><b>Step {step} of 3</b></p>
      <progress value={step} max="3" />

      {step === 1 && (<>
        <Field label="Name"  {...props('name')} />
        <Field label="Email" type="email" {...props('email')} />
      </>)}

      {step === 2 && (<>
        <Field label="Username" {...props('username')} />
        <Field label="Password" type="password" {...props('password')} />
        <Field label="Confirm password" type="password" {...props('confirm')} />
      </>)}

      {step === 3 && (<>
        <p>
          {['student', 'ta', 'instructor'].map(r => (
            <label key={r}>
              <input type="radio" name="role" value={r} checked={form.role === r} onChange={handleChange} /> {r}{' '}
            </label>
          ))}
        </p>
        <label><input type="checkbox" name="subscribe" checked={form.subscribe} onChange={handleChange} /> Subscribe to newsletter</label>
        <h4>Review</h4>
        <ul>
          <li>Name: {form.name}</li><li>Email: {form.email}</li><li>Username: {form.username}</li>
          <li>Password: {'*'.repeat(form.password.length)}</li><li>Role: {form.role}</li>
          <li>Newsletter: {form.subscribe ? 'yes' : 'no'}</li>
        </ul>
      </>)}

      <p>
        {step > 1 && <button type="button" onClick={prev}>Previous</button>}
        {step < 3 && <button type="button" onClick={next} disabled={!valid}>Next</button>}
        {step === 3 && <button>Submit</button>}
      </p>
    </form>
  );
}

import { useState } from 'react';

export default function AccessibleForm() {
  const [email, setEmail]     = useState('');
  const [touched, setTouched] = useState(false);
  const error = touched && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    ? 'Enter a valid email address' : '';

  return (
    <form noValidate onSubmit={e => { e.preventDefault(); setTouched(true); }}>
      {/* label + htmlFor/id: clicking the label focuses the input */}
      <label htmlFor="email">Email <span aria-hidden="true">(required)</span></label>
      <br />
      <input
        id="email" type="email" value={email}
        onChange={e => setEmail(e.target.value)}
        onBlur={() => setTouched(true)}
        aria-required="true"
        aria-invalid={!!error}                         // marks the invalid field
        aria-describedby={error ? 'email-error' : undefined}   // links the message
      />
      {error && <p id="email-error" role="alert" style={{ color: 'crimson' }}>{error}</p>}
      <p><button>Submit</button></p>
      <p>Test it: click the label, then tab through using only the keyboard.</p>
    </form>
  );
}

Common form mistakes

Source 11-CommonMistakes.jsx

import { useState } from 'react';

// 1. Mutating state directly: the input looks "stuck"
function MutationBug() {
  const [form, setForm] = useState({ name: '' });
  const handle = (e) => { form.name = e.target.value; setForm(form); };   // WRONG
  return <p><input value={form.name} onChange={handle} placeholder="type: nothing happens" /></p>;
}
function MutationFixed() {
  const [form, setForm] = useState({ name: '' });
  const handle = (e) => setForm(prev => ({ ...prev, name: e.target.value }));   // CORRECT
  return <p><input value={form.name} onChange={handle} placeholder="type: it works" /></p>;
}

// 2. No disabled state: double submits
function SlowForm({ protect }) {
  const [loading, setLoading] = useState(false);
  const [count, setCount]     = useState(0);
  const submit = async (e) => {
    e.preventDefault();
    setCount(c => c + 1);
    setLoading(true);
    await new Promise(r => setTimeout(r, 1500));   // slow network
    setLoading(false);
  };
  return (
    <form onSubmit={submit}>
      <button disabled={protect && loading}>{loading ? 'Sending...' : 'Send'}</button>
      <span> requests sent: {count}</span>
    </form>
  );
}

// 3. Labels: htmlFor + id
function LabelDemo() {
  return (
    <div>
      <p><label>Name (label NOT linked): </label><input id="nolink" /></p>
      <p><label htmlFor="linked">Name (label linked): </label><input id="linked" /></p>
      <small>Click each label: only the linked one focuses its input.</small>
    </div>
  );
}

// 4. Validate on blur, not only on submit
function BlurValidation() {
  const [v, setV] = useState('');
  const [touched, setTouched] = useState(false);
  const err = touched && v.trim() === '' ? 'Required' : '';
  return (
    <p><input value={v} onChange={e => setV(e.target.value)} onBlur={() => setTouched(true)} placeholder="click in, then click out" />
       {err && <span style={{ color: 'crimson' }}> {err}</span>}</p>
  );
}

export default function Demo() {
  return (
    <div>
      <h4>1. Mutating state</h4><MutationBug /><MutationFixed />
      <h4>2. Double submit (click Send twice quickly)</h4><SlowForm protect={false} /><SlowForm protect={true} />
      <h4>3. Label link</h4><LabelDemo />
      <h4>4. Validate on blur</h4><BlurValidation />
    </div>
  );
}
