React Forms & Controlled Components

HTML forms have their own state inside the DOM. React's approach is to take control of that state — making form inputs controlled components whose values are always driven by React state. This gives you a single source of truth for all form data.

1. Controlled vs Uncontrolled

Uncontrolled

  • DOM manages the value
  • Read via ref.current.value
  • Simple, less code
  • Harder to validate in real time

Controlled (preferred)

  • React state is the source of truth
  • value={state} + onChange
  • Full control over input behavior
  • Easy to validate, transform, sync
💡

Use controlled components for any form that needs real-time validation, conditional logic, or submitting values to an API. Use useRef (uncontrolled) only for simple cases like setting focus.

2. Controlled Text Inputs

A controlled input binds its value attribute to React state and updates that state via onChange.

import { useState } from 'react'; function LoginForm() { const [email, setEmail] = useState(''); const [password, setPassword] = useState(''); function handleSubmit(e) { e.preventDefault(); // prevent page reload console.log('Logging in with:', email, password); // send to API... } return ( <form onSubmit={handleSubmit}> <label> Email <input type="email" value={email} {/* controlled: tied to state */} onChange={e => setEmail(e.target.value)} {/* update state on change */} required /> </label> <label> Password <input type="password" value={password} onChange={e => setPassword(e.target.value)} minLength={8} /> </label> <button type="submit">Log in</button> </form> ); }
⚠️

Always call e.preventDefault() in onSubmit. Without it, the browser reloads the page on form submission, losing all React state.

3. Select & Textarea

Both <select> and <textarea> work the same as text inputs — bind value to state and handle onChange.

const [department, setDepartment] = useState(''); const [bio, setBio] = useState(''); // Select dropdown <select value={department} onChange={e => setDepartment(e.target.value)}> <option value="">— Select department —</option> <option value="cs">Computer Science</option> <option value="ee">Electrical Engineering</option> <option value="me">Mechanical Engineering</option> </select> {/* Textarea — also uses value, not children */} <textarea value={bio} onChange={e => setBio(e.target.value)} rows={4} placeholder="Tell us about yourself..." />
💡

In HTML, <textarea> content goes between its tags. In React, use the value prop instead — consistent with other controlled inputs.

4. Checkboxes & Radio Buttons

Checkbox — boolean state

const [agreedToTerms, setAgreedToTerms] = useState(false); <label> <input type="checkbox" checked={agreedToTerms} {/* use 'checked', not 'value' */} onChange={e => setAgreedToTerms(e.target.checked)} /> I agree to the Terms and Conditions </label>

Multiple checkboxes — array state

const skills = ['HTML', 'CSS', 'JavaScript', 'React']; const [selected, setSelected] = useState([]); function handleSkill(skill, checked) { if (checked) { setSelected(prev => [...prev, skill]); } else { setSelected(prev => prev.filter(s => s !== skill)); } } {skills.map(skill => ( <label key={skill}> <input type="checkbox" checked={selected.includes(skill)} onChange={e => handleSkill(skill, e.target.checked)} /> {skill} </label> ))} <p>Selected: {selected.join(', ')}</p>

Radio buttons — string state

const [year, setYear] = useState('1'); {['1','2','3','4'].map(y => ( <label key={y}> <input type="radio" name="year" {/* same name groups them */} value={y} checked={year === y} onChange={e => setYear(e.target.value)} /> Year {y} </label> ))}

5. Multiple Inputs — One Handler

For forms with many fields, store all values in a single state object and use e.target.name to route updates to the right field.

function RegistrationForm() { const [form, setForm] = useState({ firstName: '', lastName: '', email: '', year: '1', }); // One handler for ALL inputs — reads e.target.name to know which field function handleChange(e) { const { name, value } = e.target; setForm(prev => ({ ...prev, [name]: value })); } function handleSubmit(e) { e.preventDefault(); console.log('Submitted:', form); } return ( <form onSubmit={handleSubmit}> <input name="firstName" value={form.firstName} onChange={handleChange} placeholder="First name" /> <input name="lastName" value={form.lastName} onChange={handleChange} placeholder="Last name" /> <input name="email" value={form.email} onChange={handleChange} type="email" placeholder="Email" /> <select name="year" value={form.year} onChange={handleChange}> {['1','2','3','4'].map(y => <option key={y} value={y}>Year {y}</option>)} </select> <button type="submit">Register</button> </form> ); }
✅

Computed property name: { [name]: value } uses the variable name as the object key. This is ES6 syntax — it dynamically sets the field identified by e.target.name.

6. Form Validation

Validate on submit and show field-level error messages. Store errors in a state object keyed by field name.

function SignupForm() { const [form, setForm] = useState({ email: '', password: '' }); const [errors, setErrors] = useState({}); function validate() { const e = {}; if (!form.email) { e.email = 'Email is required'; } else if (!/\S+@\S+\.\S+/.test(form.email)) { e.email = 'Enter a valid email address'; } if (!form.password) { e.password = 'Password is required'; } else if (form.password.length < 8) { e.password = 'Password must be at least 8 characters'; } return e; } function handleChange(e) { const { name, value } = e.target; setForm(prev => ({ ...prev, [name]: value })); // Clear error for this field as the user types if (errors[name]) setErrors(prev => ({ ...prev, [name]: '' })); } function handleSubmit(e) { e.preventDefault(); const newErrors = validate(); if (Object.keys(newErrors).length > 0) { setErrors(newErrors); return; } // All valid — submit alert('Account created!'); } return ( <form onSubmit={handleSubmit}> <div> <input name="email" value={form.email} onChange={handleChange} type="email" placeholder="Email" style={{ borderColor: errors.email ? 'red' : '' }} /> {errors.email && <p style={{color:'red',fontSize:13}}>{errors.email}</p>} </div> <div> <input name="password" value={form.password} onChange={handleChange} type="password" placeholder="Password" style={{ borderColor: errors.password ? 'red' : '' }} /> {errors.password && <p style={{color:'red',fontSize:13}}>{errors.password}</p>} </div> <button type="submit">Create account</button> </form> ); }

function ContactForm() { const [form, setForm] = useState({ name: '', email: '', message: '' }); const [errors, setErrors] = useState({}); const [submitted, setSubmitted] = useState(false); function handleChange(e) { const { name, value } = e.target; setForm(p => ({ ...p, [name]: value })); setErrors(p => ({ ...p, [name]: '' })); } function handleSubmit(e) { e.preventDefault(); const errs = {}; if (!form.name.trim()) errs.name = 'Name is required'; if (!form.email.trim()) errs.email = 'Email is required'; if (!form.message.trim()) errs.message = 'Message is required'; if (Object.keys(errs).length) { setErrors(errs); return; } setSubmitted(true); } if (submitted) return <p>Thank you, {form.name}! We'll be in touch.</p>; return ( <form onSubmit={handleSubmit}> {['name','email','message'].map(field => ( <div key={field}> {field === 'message' ? <textarea name={field} value={form[field]} onChange={handleChange} rows={4} /> : <input name={field} value={form[field]} onChange={handleChange} /> } {errors[field] && <p style={{color:'red',fontSize:13}}>{errors[field]}</p>} </div> ))} <button type="submit">Send</button> </form> ); }

const SKILLS = ['HTML','CSS','JavaScript','React','Node.js','Python']; function SkillPicker() { const [selected, setSelected] = useState([]); function toggle(skill, checked) { setSelected(prev => checked ? [...prev, skill] : prev.filter(s => s !== skill)); } return ( <div> <p>Selected: {selected.length} / 3</p> {selected.length > 3 && <p style={{color:'red'}}>Please select at most 3 skills.</p>} {SKILLS.map(skill => ( <label key={skill}> <input type="checkbox" checked={selected.includes(skill)} onChange={e => toggle(skill, e.target.checked)} /> {skill} </label> ))} </div> ); }
